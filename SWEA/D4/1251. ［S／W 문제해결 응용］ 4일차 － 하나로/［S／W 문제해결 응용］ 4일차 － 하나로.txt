
import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.Arrays;
import java.util.StringTokenizer;

public class Solution{

    public static void main(String[] args) throws IOException {

        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));

        int T = Integer.parseInt(br.readLine());

        for (int tc = 1; tc <= T; tc++) {

            int N = Integer.parseInt(br.readLine());

            int[] cords_x = new int[N];
            int[] cords_y = new int[N];

            StringTokenizer st = new StringTokenizer(br.readLine());

            for (int i = 0; i < N; i++) {
                cords_x[i] = Integer.parseInt(st.nextToken());
            }

            st = new StringTokenizer(br.readLine());

            for (int i = 0; i < N; i++) {
                cords_y[i] = Integer.parseInt(st.nextToken());
            }

            double E = Double.parseDouble(br.readLine());

            boolean[] visited = new boolean[N];
            double[] minEdge = new double[N];

            Arrays.fill(minEdge, Double.MAX_VALUE);

            minEdge[0] = 0;

            double result = 0;

            for (int i = 0; i < N; i++) {

                double min = Double.MAX_VALUE;
                int minVertex = -1;

                for (int j = 0; j < N; j++) {

                    if (!visited[j] && minEdge[j] < min) {
                        min = minEdge[j];
                        minVertex = j;
                    }
                }

                visited[minVertex] = true;

                result += min;

                for (int j = 0; j < N; j++) {

                    if (!visited[j]) {

                        long dx = (long) cords_x[minVertex] - cords_x[j];
                        long dy = (long) cords_y[minVertex] - cords_y[j];

                        double distance = dx * dx + dy * dy;
                        double weight = distance * E;

                        if (weight < minEdge[j]) {
                            minEdge[j] = weight;
                        }
                    }
                }
            }

            System.out.println("#" + tc + " " + Math.round(result));
        }
    }
}
